2. 通信协议

写入格式

{method}|{data}

示例：queryPortName|（无参）或 sendOrder|{"SampleID":"S001",...}

读取格式

{保留字段}|{状态码}|{数据}

BaseResponse<string> 通用返回字段

字段类型说明
IsSuccessbool通信是否成功（含业务失败时也可能为 true）
ResultCodeint1=成功，0=业务失败，-1=异常/连接失败
ResultMessageCodestring成功时为 0x0000000000000000；失败时为 16 位 hex 错误码
ResultMessagestring可读消息（部分接口填充）
Datastring业务数据，通常为 JSON 字符串，需二次反序列化

状态码转换规则（ConvertBaseResponse）

管道返回状态码ResultCodeResultMessageCode
"0"10x0000000000000000
非 "0" 数字00x{16位hex}
解析异常 / 连接异常-1SE500145

3. 调用指南

PipeConnectionHelper 类源码

public class PipeConnectionHelper
{
    public BaseResponse<string> PipeSend(ClientPipeConnection clientConnection, string method, string data)
    {
        var response = new BaseResponse<string>();
        var _logger = new LoggerService();
        try
        {
            clientConnection.Connect();
            clientConnection.Write($"{method}|{data}");
            response = ConvertBaseResponse(clientConnection.Read());
            clientConnection.Close();
        }
        catch (Exception ex)
        {
            response.IsSuccess = false;
            response.ResultCode = -1;
            response.ResultMessageCode = EnumMsgCode.SE500145.ToString();
            response.Data = "";
            clientConnection.Dispose();
            _logger.Error(ex, "Pipe Send Error");
            return response;
        }
        return response;
    }

    private BaseResponse<string> ConvertBaseResponse(string namePipeResult)
    {
        var _logger = new LoggerService();
        var response = new BaseResponse<string>();
        _logger.Info($"======namePipeResult====== {namePipeResult}");
        try
        {
            var namePipeResults = namePipeResult.Split('|');
            if (namePipeResults[1] == "0")
            {
                response.IsSuccess = true;
                response.ResultCode = 1;
                response.ResultMessageCode = "0x0000000000000000";
                response.Data = namePipeResults[2].ToString();
            }
            else
            {
                response.IsSuccess = true;
                response.ResultCode = 0;
                response.ResultMessageCode = String.Format("0x{0:x16}", long.Parse(namePipeResults[1]));
                response.Data = namePipeResults[2].ToString();
            }
        }
        catch (Exception ex)
        {
            response.IsSuccess = false;
            response.ResultCode = -1;
            _logger.Error(ex, "Convert Base Response Error");
        }
        return response;
    }
}

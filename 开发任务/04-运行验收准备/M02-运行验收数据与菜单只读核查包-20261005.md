--    提供方只读 res_culture，res_batch 仅 LEFT JOIN 核归属；SQL 内不判态、不 limit、不 FOR UPDATE

       OR lifecycle_state NOT IN
          ('DRAFT','NOT_STARTED','RUNNING','EXPIRED','DELAYED','CANCELLED','FINISHED'));

SHOW CREATE TABLE res_culture;

-- B. 实验室靶子可用性（workflow 行 + lab_type 默认 '' 共同锁死 M02-236）
SELECT l.lab_id, l.lab_type, l.status, l.del_flag, l.ip_address, l.port,

-- C. 守卫基线
SELECT scope_key, revision, update_time FROM sys_management_revision WHERE scope_key = 'SYSTEM';

WHERE del_flag = '0';

SELECT user_id, lab_id, auth_type FROM sys_user_lab WHERE lab_id = 9301;

-- E. OBS 键与演示错标绑定行
SELECT mapping_id, device_kind, device_id, channel_code, raw_metric_code,
       mapping_version, item_id

SELECT detector_id AS id, 'DETECTOR' AS kind, lab_id, del_flag
FROM lab_detector WHERE detector_id = 9601
UNION ALL SELECT scale_id, 'FLOOR_SCALE', lab_id, del_flag
FROM lab_floor_scale WHERE scale_id = 9965
UNION ALL SELECT device_id, 'DOWNSTREAM_DEVICE', lab_id, del_flag
FROM lab_downstream_device WHERE device_id = 9964;

SELECT item_id, detect_class, item_code, del_flag FROM dev_detector_item WHERE item_id = 9602;

-- F. 主数据引用（236 在守卫事务内重检，失效即 INPUT_INVALID/400，不是业务冲突码）

## 5. 阻塞汇总

### BLOCKED_DATA（本批新增两条硬阻塞）

- **D-新1 · 安装链无可登录主体**：`docs/开发基线/02-基础数据初始化.sql` 全文只插 `apr_scene`、`sys_config`、`sys_management_revision` 三张表，**零 `sys_user`/`sys_role`/`sys_menu`**。而 `UserConstants.isAdmin(userId)` 定义为 `userId != null && 1L == userId`，`SysPermissionServiceImpl#getMenuPermission` 的超管旁路（塞 `*:*:*`）同样只在 `user.isAdmin()` 时触发 ⇒ **仓库内没有任何 user_id=1 的种子**。唯一账号来自 `05-开发调试测试数据.sql` 的 9001，其 `password` 走 DDL 默认 `''`。
  ⇒ 六接口 HTTP/浏览器验收在"起步主体"这一步就不可执行；**"直接改用超管账号"这一常见建议本身也需要先补账号种子**（见 §10 交叉引用），不得默认超管可用。
- **D-新2 · lab 权限菜单/种子缺口**：`lab:observation:maintain`、`lab:observation:query` 在**全仓所有 `.sql` 中零命中**，含旧库导出 `sql/sys_menu.sql`（该文件有 355 条菜单 INSERT、含 `lab:lab:{query,edit,remove}` 各 1 行，但观测码一条都没有）。且 `docs/开发基线/README.md` 明确把 `sql/ymsw.sql` 一类定性为"只用于提取白名单表结构，绝不直接执行其旧库导出/业务 INSERT" ⇒ 旧导出不构成可安装种子。
- 沿用附录 §8 的第 1/2/3/4/5/11/12/13 项：`res_culture` 事实组、`del_flag` NULL 三值逻辑坑、培养归属一致性、READY 的四条连带 CHECK、"干净靶子需无授权∧无负责人"、venue/dept 主数据、`lab_workflow_node` + `lab_type` 锁死 236。

CREATE TABLE sys_menu (

    menu_id BIGINT AUTO_INCREMENT PRIMARY KEY COMMENT '权限ID',

    parent_id BIGINT DEFAULT 0 COMMENT '父级权限ID，0=顶层模块',

    menu_name VARCHAR(64) NOT NULL COMMENT '权限展示名称（前端菜单显示文字）',

    menu_code VARCHAR(100) NOT NULL UNIQUE COMMENT '权限编码（全局唯一）',

    menu_type TINYINT NOT NULL COMMENT '节点类型：1系统模块 2页面菜单 3按钮/操作',

    sort INT DEFAULT 0 COMMENT '树形排序，数字越小越靠前',

    icon VARCHAR(128) DEFAULT NULL COMMENT '前端菜单图标',

    route_path VARCHAR(200) DEFAULT NULL COMMENT '前端路由地址（仅页面类型生效）',

    component VARCHAR(200) DEFAULT NULL COMMENT '前端页面组件路径（仅页面类型生效）',

    permission_api VARCHAR(255) DEFAULT NULL COMMENT '绑定后端接口地址，多个逗号分隔（按钮/页面鉴权用）',

    visible TINYINT DEFAULT 1 COMMENT '是否在菜单栏展示：0隐藏 1显示（按钮节点统一填0）',

    status TINYINT DEFAULT 1 COMMENT '权限状态：0停用 1启用，停用后所有角色无法访问',

    remark VARCHAR(255) DEFAULT NULL COMMENT '备注：记录这个功能是哪个模块新增、用途',

    create_time DATETIME DEFAULT NOW(),

    update_time DATETIME DEFAULT NOW(),

    create_user_id BIGINT NOT NULL,

    update_user_id BIGINT NULL,

    UNIQUE KEY uk_menu_code (menu_code)

) COMMENT='功能菜单/按钮权限表，权限树数据源';
